// unauthenticated control that must answer 401 or 403. Writes probes.json.

const b64u = (b) => Buffer.from(b).toString('base64url');
function jwt(sub) {
  const now = Math.floor(Date.now() / 1000);
  const h = b64u(JSON.stringify({ alg: 'HS256', typ: 'JWT' }));
  const p = b64u(JSON.stringify({ sub, type: 'access', iat: now, exp: now + 3600 }));
  const s = crypto.createHmac('sha256', SECRET).update(`${h}.${p}`).digest('base64url');
  return `${h}.${p}.${s}`;
}
const ownerJwt = jwt(OWNER);

const probes = [];
const routes = {}; // entity -> [probe ids]
let seq = 0;

async function call(method, path, token, body) {
  const res = await fetch(BASE + path, {
    method,
    headers: {
      ...(token ? { Authorization: `Bearer ${token}` } : {}),
      ...(body !== undefined ? { 'Content-Type': 'application/json' } : {}),
    },
    body: body !== undefined ? JSON.stringify(body) : undefined,
  });
  const text = await res.text();
  let json = null;
  try { json = JSON.parse(text); } catch { /* not json */ }
  return { status: res.status, text, json };
}

/** One route = an authenticated probe + an anonymous control. */
async function probe(entity, method, path, { token = ownerJwt, body, expect, who = 'owner', anonPath } = {}) {
  const r = await call(method, path, token, body);
  const id = `${who}-${String(++seq).padStart(3, '0')}`;
  const ok = (Array.isArray(expect) ? expect : [expect]).includes(r.status);
  probes.push({
    id, kind: 'http', executed: true, method, path, status: r.status,
    outcome: ok ? 'verified' : 'failed',
    reason: `real candidate-local request as ${who}; expected ${[].concat(expect).join('|')}, received ${r.status}`,
    response_sha256: crypto.createHash('sha256').update(r.text).digest('hex'),
    response_bytes: Buffer.byteLength(r.text),
  });
  if (!ok) console.error(`FAIL ${entity} ${r.status} ${r.text.slice(0, 300)}`);
  const a = await call(method, anonPath ?? path, null, body);
  const aid = `anonymous-${String(++seq).padStart(3, '0')}`;
  const aok = a.status === 401 || a.status === 403;
  probes.push({
    id: aid, kind: 'http', executed: true, method, path: anonPath ?? path, status: a.status,
    outcome: aok ? 'verified' : 'failed',
    reason: `unauthenticated control expected 401 or 403; expected 401|403, received ${a.status}`,
    response_sha256: crypto.createHash('sha256').update(a.text).digest('hex'),
    response_bytes: Buffer.byteLength(a.text),
  });
  if (!aok) console.error(`FAIL-ANON ${entity} ${a.status} ${a.text.slice(0, 200)}`);
  routes[entity] = [id, aid];
  return r;
}
