#!/bin/bash
# usage: run_sandbox.sh <repo-worktree> <canary-record-dir> <plan-dir> <out-dir>
# One sandbox: private user/net/pid/ipc/uts namespaces, source read-only at /repo, record directory read-only at
# /canary, plan read-only at /plan, output at /out. No network. python3 and its stdlib come from the read-only /usr.
set -eu
WT=$1; CD=$2; PD=$3; OD=$4
exec bwrap --unshare-user --unshare-net --unshare-pid --unshare-ipc --unshare-uts --die-with-parent \
  --tmpfs / --ro-bind /usr /usr --symlink usr/lib /lib --symlink usr/lib64 /lib64 --symlink usr/bin /bin \
  --proc /proc --dev /dev --tmpfs /tmp --dir /home/canary \
  --ro-bind "$WT" /repo --ro-bind "$CD" /canary --ro-bind "$PD" /plan --bind "$OD" /out \
  --setenv CI 1 --setenv NO_COLOR 1 --setenv HOME /home/canary --setenv LANG C.UTF-8 --setenv PATH /usr/bin:/bin \
  --setenv TMPDIR /tmp --setenv TZ UTC --setenv PYTHONDONTWRITEBYTECODE 1 \
  /usr/bin/python3 -I /canary/harness.py.txt
